# Names of every reviewed benchmark
curl -s https://awesomecua.com/downloads/catalog.jsonl \
  | jq -r 'select(.kind=="benchmark" and .review_status=="curated") | .name'

# Reviewed papers as BibTeX
curl -sO https://awesomecua.com/downloads/citations.bib

# Query the catalog with SQL
curl -sO https://awesomecua.com/downloads/catalog.sqlite
sqlite3 catalog.sqlite "select name from resources where kind='dataset' limit 5"
import json, urllib.request

url = "https://awesomecua.com/downloads/catalog.jsonl"
rows = [json.loads(line) for line in urllib.request.urlopen(url)]
reviewed = [r for r in rows if r["review_status"] == "curated"]
print(len(reviewed), reviewed[0]["name"])
from datasets import load_dataset

ds = load_dataset("TIGER-Lab/AwesomeCUA", "resources", split="train")
benchmarks = ds.filter(lambda r: r["kind"] == "benchmark")
print(benchmarks[0]["name"], benchmarks[0]["url"])
# One retrieval document: its text and metadata fields
curl -s https://awesomecua.com/downloads/rag-documents.jsonl \
  | head -n 1 | jq '{id, text: .text[:120], metadata: (.metadata | keys)}'
